\begin{frame}[t]%
\frametitle{Types of Metaheuristics}%
%
\begin{itemize}%
%
\item We can build many interesting algorithms just using such components~(and sometimes a bit more problem-specific information).
%
\item<2-> Local Search\only<3->{:%
\begin{itemize}%
\item Keep one solution~$x_c$, sample at least one new solution~$x_n$ from its neighborhood in each step, move to better solutions as discovered.%
%
\only<-5>{%
\item<4-> {\color{green!60!black}Advantage}: Fast%
\item<5-> {\color{red!60!black}Disadvantage}: May get stuck at a local optimum%
}%
\end{itemize}%
}%
%
\item<6-> Population-based Methods\only<7->{:%
\begin{itemize}%
\item Maintain population~$S$ of $|S|=\mu\geq1$ solutions, sample set~$N$ of $|N|=\lambda\geq1$ new solutions in each step by creating modified copies of the elements in~$S$, then update $S$.%
\only<-9>{%
\item<8-> {\color{green!60!black}Advantage}: Less likely to get stuck at local optimum.%
\item<9-> {\color{red!60!black}Disadvantage}: Slow.%
}%
\end{itemize}%
}%
%
\item<10-> Model-based Methods\only<11->{:%
\begin{itemize}%
\item Maintain a stochastic model~$M$ (usually a multivariate probility distribution) of what good solutions could look like, sample set~$N$ of $|N|=\lambda\geq1$ new solutions based on~$M$ in each step, then update~$M$.%
\only<-13>{%
\item<12-> {\color{green!60!black}Advantage}: Sometimes really good performance.%
\item<13-> {\color{red!60!black}Disadvantage}: Complicated, sampling and model update may be slow.%
}%
\end{itemize}%
}%
%
\item<14-> Hybrid Algorithms\only<15->{:%
\begin{itemize}%
\item Usually combine model-based or population-based approach with local search: Each new solution from model/population is refined with local search.%
\only<-17>{%
\item<16-> {\color{green!60!black}Advantage}: Fast (because local search), good global search (because of population/model)%
\item<17-> {\color{red!60!black}Disadvantage}: usually few.%
}%
\end{itemize}%
}%
%
\end{itemize}%
%
\end{frame}%

\section{Fields where Metaheuristics can be Applied}%
%
\begin{frame}%
\frametitle{Types of Search Spaces / Decision Variables}%
%
\begin{itemize}%
\item Discrete problems have solutions that can be expressed as bit strings of length~$n$, i.e., search in~$\{0,1\}^n$.%
%
\item<2-> Combinatorial problems have solutions that can be expressed as permutations of first $n$~natural numbers.%
%
\item<3-> Integer problems have solutions that can be expressed as integer vectors~$\subset\integerNumbers^n$.%
%
\item<4-> Continuous problems have solutions that can be expressed as real vectors~$\subset\realNumbers^n$.%
%
\item<5-> Mixed problems: usually discrete + integer + continuous.%
%
\item<6-> Genetic Programming usually has solutions that can be expressed as tree data structures with different types of nodes and a maximum depth~(imagine mathematical expressions like~$(5x+3)/3$, which have inherent tree nature).
\end{itemize}%
\end{frame}%
%
\begin{frame}%
\frametitle{Mode of the Objective Function(s)}%
%
\begin{itemize}%
\item The objective function~$f(x)$ tells us how good a solution~$x$ is; the smaller~$f(x)$, the better.%
%
\item<2-> $f$~can be a mathematical function with properties such as differentiability, etc.%
%
\item<3-> $f$~can be discrete function, like on the TSP.%
%
\item<4-> $f$~may require us to run complex simulations, e.g., if $x$~is some engineering design or a policy that needs to be simulated.%
%
\item<5-> $f$~usually is static and does not change{\dots}%
%
\item<6-> {\dots}but sometimes, $f$~is dynamic and may change while we try to optimize it.%
%
\item<7-> Sometimes, we face a multi-objective problem, where we do not have a single objective function, but multiple functions, i.e., a vector function~$\vec{f}(x)$, and we try to trade-off different goals; for example speed vs.\ cost.%
\end{itemize}%
\end{frame}%
